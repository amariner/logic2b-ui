## Uso

```tsx
import {
  Empty,
  EmptyContent,
  EmptyDescription,
  EmptyHeader,
  EmptyMedia,
  EmptyTitle,
} from "@/components/ui/empty"

<Empty>
  <EmptyHeader>
    <EmptyMedia variant="icon">
      <InboxIcon />
    </EmptyMedia>
    <EmptyTitle>Todavía no hay mensajes</EmptyTitle>
    <EmptyDescription>
      Cuando recibas un mensaje, aparecerá aquí.
    </EmptyDescription>
  </EmptyHeader>
  <EmptyContent>
    <Button size="sm">Redactar mensaje</Button>
  </EmptyContent>
</Empty>
```
